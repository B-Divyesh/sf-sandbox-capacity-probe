staging-west / surveyscp 0.1.0
$ cargo install --path cli
$ scp probe \
  --target staging-west \
  --confirm staging-west \
  --containers 12 \
  --ports-per-container 2 \
  --mounts 1 \
  --samples 3 \
  --output capacity.json
